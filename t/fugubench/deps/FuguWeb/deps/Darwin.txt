# The runtime dist entry takes the signify tier, and scripts/deps
# needs signify(1) to check that signature. The tool environment
# installs before every other one, so the command is there when the
# runtime chain reaches the entry. t/fuguweb/rotate.t uses the same
# command, and deps-test depends on deps.
tool pkg signify-osx

# The secret gate of make check runs gitleaks, so the tool
# environment installs it before every other environment.
tool bin gitleaks https://github.com/gitleaks/gitleaks/releases/download/v8.30.1/gitleaks_8.30.1_{os}_{arch}.tar.gz gitleaks

runtime dist https://github.com/FuguBSD/Fugu/releases/download/v0.5.0/Fugu.tar.gz
runtime pkg mandoc
runtime pkg lowdown

# gpg(1) makes an OpenPGP key and its binding, per WEB-OPENPGP.
# t/fuguweb/rotate.t and t/fuguweb/keys.t each skip the assertions
# that need it.
runtime pkg gnupg
