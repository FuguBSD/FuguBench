# External dependencies of the website. scripts/deps installs each
# line. fuguweb rotate-key signs the key manifest, so it needs
# signify(1). The site build itself signs nothing.
tool pkg signify-openbsd
# The secret gate of make check runs gitleaks, so the tool environment
# installs it before every other environment.
tool bin gitleaks https://github.com/gitleaks/gitleaks/releases/download/v8.30.1/gitleaks_8.30.1_{os}_{arch}.tar.gz gitleaks

# fuguweb rotate-key runs in the rotation workflow, beside a private
# key, so each version is pinned here and never resolved at run time
# (SITE-ROTATE-15). Each release carries a signed SHA256 manifest, so
# both entries take the signify tier: scripts/deps verifies the
# manifest with the declared key before cpanm reads the tarball.
runtime dist https://github.com/FuguBSD/Fugu/releases/download/v0.4.0/Fugu.tar.gz
runtime dist https://github.com/FuguBSD/FuguWeb/releases/download/v0.5.0/App-FuguWeb.tar.gz
runtime pkg mandoc
runtime pkg lowdown
