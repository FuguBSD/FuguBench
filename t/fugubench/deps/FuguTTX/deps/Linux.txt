# External dependencies of FuguTTX. scripts/deps installs each line.

# The runtime dist entry takes the signify tier, and scripts/deps
# needs signify(1) to check that signature. The tool environment
# installs before every other one, so the command is there when the
# runtime chain reaches the entry.
tool pkg signify-openbsd

# The secret gate of make check runs gitleaks, so the tool
# environment installs it before every other environment.
tool bin gitleaks https://github.com/gitleaks/gitleaks/releases/download/v8.30.1/gitleaks_8.30.1_{os}_{arch}.tar.gz gitleaks

# The Fugu distribution supplies the harness modules (D7).
runtime dist https://github.com/FuguBSD/Fugu/releases/latest/download/Fugu.tar.gz
# The Scaleway CLI operates the infrastructure (spec/infrastructure.md).
runtime bin scw https://github.com/scaleway/scaleway-cli/releases/download/v2.61.0/scaleway-cli_2.61.0_{os}_{arch}
